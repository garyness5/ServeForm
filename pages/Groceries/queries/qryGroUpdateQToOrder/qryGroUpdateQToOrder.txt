set
    to_order = {{ appsmith.store.gro_queue_to_order === true }},
    updated_at = now()
where id = {{ Number(appsmith.store.gro_queue_row_id || 0) }}
  and client_id = '{{ appsmith.store.current_client_id }}'::uuid

    event_id,
