select
    status_value as value,
    display_label as label
from public.system_status_options
where domain_code = 'event'
  and active = true
  and status_value in (
      'Draft',
      'Accepted',
      'Ordered',
      'Closed',
      'Cancelled'
  )
order by
    case status_value
        when 'Draft' then 1
        when 'Accepted' then 2
        when 'Ordered' then 3
        when 'Closed' then 4
        when 'Cancelled' then 5
        else 99
    end;
